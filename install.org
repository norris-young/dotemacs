* how to install emacs in an online distro in wsl
** install wsl
#+begin_src bat
  wsl --install
  wsl --update
#+end_src
** install a wsl distro
Ubuntu distro might already be installed in the previous step, or you can install ubuntu with
following instructions.
#+begin_src bat
  :: get online distro versions
  wsl --list --online
  :: then install the distro you need, e.g. Ubuntu-22.04
  wsl --install Ubuntu-22.04
  wsl --install Ubuntu-20.04
#+end_src
** clone emacs repos
#+begin_src bash
  git clone https://github.com/emacs-mirror/emacs.git
#+end_src
** install emacs build dependencies
*** build tools
#+begin_src bash
  # install ubuntu build tools
  sudo apt install build-essential autoconf cmake
#+end_src
*** gtk
#+begin_src bash
  sudo apt install libgtk-3-dev libglib2.0-dev
#+end_src
*** gcc jit
**** ubuntu 22.04
#+begin_src bash
  sudo apt install libgccjit-11-dev
#+end_src
**** ubuntu 20.04
#+begin_src bash
  sudo apt install libgccjit-9-dev
#+end_src
*** tree-sitter
**** online
#+begin_src bash
  sudo apt install libtree-sitter-dev
#+end_src
**** offline
***** add local PATH
#+begin_src bash
  export PATH=$HOME/usr/local/bin:$PATH
  export LD_LIBRARY_PATH=$HOME/usr/local/lib:$LD_LIBRARY_PATH
#+end_src

***** install rust(optional)
download offline installer of rust from https://forge.rust-lang.org/infra/other-installation-methods.html
#+begin_src bash
  tar xvf rust-1.93.1-x86_64-unknown-linux-gnu.tar.xz
  cd rust-1.93.1-x86_64-unknown-linux-gnu/
  ./install.sh --prefix=$HOME/usr/local
#+end_src


***** install tree-sitter
#+begin_src bash
  git clone https://github.com/tree-sitter/tree-sitter.git
  # checkout latest branch
  git checkout -b release-0.26 origin/release-0.26
  # build for release-0.26
  mkdir build
  cd build
  cmake -DCMAKE_INSTALL_PREFIX=$HOME/usr/local ..
  make -j
  make install
  # checkout compatible older branch, e.g. emacs-30::release-0.25
  git checkout -b release-0.25 origin/release-0.25
  # build for release-0.25
  make -j
  PREFIX=$HOME/usr/local make install
#+end_src

** configure git
#+begin_src bash
  git config --global user.emacs "a1ph2@outlook.com"
  git config --global user.name "a1ph2"
#+end_src

** build emacs

*** online
#+begin_src bash
  # checkout latest release branch
  git checkout -b emacs30 origin/emacs-30
  # cherry pick my whitespace mode fix
  git fetch https://github.com/norris-young/emacs.git 7a2bcc92b40e6e5acd3d7e53694720bebc37b16b && git cherry-pick FETCH_HEAD
  # build instructions
  ./autogen.sh
  CFLAGS="-march=native -O2" ./configure --with-pgtk --with-native-compilation=aot --with-tree-sitter --without-compress-install
  make -j
#+end_src

*** offline
#+begin_src bash
  # clone ssh local repo
  git clone localclient:~/repos/emacs/.git
  # build instructions
  ./autogen.sh
  PKG_CONFIG_PATH=$HOME/usr/local/lib/pkgconfig/ CFLAGS="-march=native -O2" ./configure --with-pgtk --with-native-compilation=aot --with-tree-sitter --without-compress-install --prefix=$HOME/usr/local
  make -j
#+end_src

** clone my personal emacs config
#+begin_src bash
  git clone https://github.com/norris-young/dotemacs.git
  ln -s dotemacs .emacs.d
#+end_src

** install emacs config dependencies
*** ctags
~this package should be installed after emacs, since emacs will override ctags binary~
#+begin_src bash
  sudo rm /usr/local/bin/ctags
  sudo apt install universal-ctags
#+end_src

*** gtags--gnu global
#+begin_src bash
  sudo apt install global
#+end_src

**** python
#+begin_src bash
  sudo apt install python3 python3-pip python-is-python3
  python -m pip install --upgrade pip
#+end_src

**** pygments
#+begin_src bash
  pip install pygments
#+end_src

**** configuration
#+begin_src bash
  cp /etc/gtags/gtags.conf $HOME/.config/global/gtags.conf
  export GTAGSCONF=$HOME/.config/global/gtags.conf
  export GTAGSLABEL=pygments
#+end_src

*** lsp bridge
**** python dependencies
#+begin_src bash
  lsp_bridge_pyenv_path=~/.emacs.d/lsp.bridge.pyenv
  python -m venv --system-site-packages ${lsp_bridge_pyenv_path}
  source ${lsp_bridge_pyenv_path}/bin/activate
  pip install epc orjson sexpdata six setuptools paramiko rapidfuzz watchdog packaging
#+end_src

***** clangd
****** install prebuilt package
#+begin_src bash
  # get ubunutu source from llvm project https://apt.llvm.org/
  deb http://apt.llvm.org/jammy/ llvm-toolchain-jammy-22 main
  # install latest llvm version
  sudo apt install clangd-22
#+end_src
****** build from source
#+begin_src bash
  git clone https://github.com/llvm/llvm-project.git
  # checkout lastes release branch
  git checkout -b 22.x origin/release/22.x
  # build
  mkdir build
  cd build
  cmake ../llvm -DLLVM_ENABLE_PROJECTS="clang;clang-tools-extra" -DCMAKE_BUILD_TYPE=Release
  cmake --build .
  # install for system
  cmake --build . --target install
  # install for local user
  cmake -DCMAKE_INSTALL_PREFIX=$HOME/usr/local -P cmake_install.cmake
#+end_src
